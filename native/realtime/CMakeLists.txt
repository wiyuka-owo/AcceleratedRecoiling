cmake_minimum_required(VERSION 3.16)
project(AcceleratedRecoilingRealtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT AR_JNI_HEADERS)
    message(FATAL_ERROR "Set AR_JNI_HEADERS to a JDK include directory or the project's bundled JNI headers")
endif()
add_library(AcceleratedRecoilingRealtime SHARED query.cpp batch_kernel.cpp)
option(AR_KERNEL_TESTS "Build standalone SIMD differential tests" OFF)
if(AR_KERNEL_TESTS)
    add_executable(RealtimeKernelTest batch_kernel_test.cpp batch_kernel.cpp)
    if(WIN32 OR AR_WINDOWS)
        target_compile_options(RealtimeKernelTest PRIVATE /O2 /EHsc /fp:precise /MT)
    else()
        target_compile_options(RealtimeKernelTest PRIVATE -O3 -fno-fast-math)
    endif()
endif()
target_include_directories(AcceleratedRecoilingRealtime PRIVATE "${AR_JNI_HEADERS}")
if(WIN32 OR AR_WINDOWS)
    target_include_directories(AcceleratedRecoilingRealtime PRIVATE "${AR_JNI_HEADERS}/win32")
    target_compile_options(AcceleratedRecoilingRealtime PRIVATE /O2 /EHsc /fp:precise /MT)
elseif(APPLE OR AR_DARWIN)
    target_include_directories(AcceleratedRecoilingRealtime PRIVATE "${AR_JNI_HEADERS}/darwin")
    target_compile_options(AcceleratedRecoilingRealtime PRIVATE -O3 -fno-fast-math)
else()
    target_include_directories(AcceleratedRecoilingRealtime PRIVATE "${AR_JNI_HEADERS}/linux")
    target_compile_options(AcceleratedRecoilingRealtime PRIVATE -O3 -fno-fast-math)
    target_link_options(AcceleratedRecoilingRealtime PRIVATE -static-libstdc++ -static-libgcc -Wl,--no-undefined)
endif()
